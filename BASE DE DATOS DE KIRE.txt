BASE DE DATOS 
USE kire_crepas;
CREATE TABLE IF NOT EXISTS bases (
  id_base INT AUTO_INCREMENT PRIMARY KEY,
  nombre VARCHAR(50) NOT NULL,
  tipo ENUM('DULCE', 'SALADA') NOT NULL DEFAULT 'DULCE',
  activo BOOLEAN DEFAULT TRUE,
  disponible BOOLEAN NOT NULL DEFAULT TRUE
);

CREATE TABLE IF NOT EXISTS frutas (
  id_fruta INT AUTO_INCREMENT PRIMARY KEY,
  nombre VARCHAR(50) NOT NULL,
  activo BOOLEAN DEFAULT TRUE,
  disponible BOOLEAN NOT NULL DEFAULT TRUE
);
CREATE TABLE IF NOT EXISTS pedidos (
  id_pedido INT AUTO_INCREMENT PRIMARY KEY,
  fecha TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  total DECIMAL(10, 2) NOT NULL,
  efectivo DECIMAL(10, 2) NOT NULL,
  cambio DECIMAL(10, 2),
  tipo_entrega VARCHAR(20) NOT NULL DEFAULT 'LOCAL',
  costo_envio DECIMAL(10, 2) NOT NULL DEFAULT 0.00,
  estado VARCHAR(20) DEFAULT 'ACTIVO',
  observaciones VARCHAR(500) NULL,
  id_usuario INT,
  id_corte INT NULL,

  FOREIGN KEY (id_usuario)
    REFERENCES usuarios(id_usuario) ON DELETE SET NULL,

  CONSTRAINT fk_pedidos_corte
    FOREIGN KEY (id_corte)
    REFERENCES cortes_caja(id_corte)
);

CREATE TABLE IF NOT EXISTS detalles_pedido (
    id_detalle INT AUTO_INCREMENT PRIMARY KEY,
    id_pedido INT NOT NULL,
    categoria VARCHAR(50) DEFAULT 'Sin categoría',
    producto VARCHAR(200) NOT NULL,
    cantidad INT NOT NULL,
    subtotal DECIMAL(10,2) NOT NULL,

    FOREIGN KEY (id_pedido)
        REFERENCES pedidos(id_pedido)
        ON DELETE CASCADE
);

CREATE TABLE IF NOT EXISTS cortes_caja (
    id_corte INT AUTO_INCREMENT PRIMARY KEY,
    fecha_apertura TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    fecha_cierre TIMESTAMP NULL,
    id_usuario_apertura INT,
    id_usuario_cierre INT,
    total_ventas DECIMAL(10,2),
    total_efectivo DECIMAL(10,2),
    total_cambio DECIMAL(10,2),
    total_pedidos INT,
    total_productos INT,
    estado VARCHAR(20) DEFAULT 'ABIERTO',
    observaciones TEXT,
    FOREIGN KEY (id_usuario_apertura) REFERENCES usuarios(id_usuario),
    FOREIGN KEY (id_usuario_cierre) REFERENCES usuarios(id_usuario)
);
----------------------------------------------------------------------------------------------

CREATE TABLE IF NOT EXISTS categorias (
    id_categoria INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(50) NOT NULL UNIQUE,
    es_especial BOOLEAN DEFAULT FALSE,
    activo BOOLEAN DEFAULT TRUE,
    disponible BOOLEAN NOT NULL DEFAULT TRUE
);

CREATE TABLE IF NOT EXISTS productos (
    id_producto INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL,
    precio DECIMAL(10,2) NOT NULL,
    imagen VARCHAR(255) DEFAULT 'sin_imagen.png',
    id_categoria INT NOT NULL,
    activo BOOLEAN DEFAULT TRUE,
    disponible BOOLEAN NOT NULL DEFAULT TRUE,
    FOREIGN KEY (id_categoria) REFERENCES categorias(id_categoria)
);


CREATE TABLE IF NOT EXISTS extras (
    id_extra INT AUTO_INCREMENT PRIMARY KEY,
    id_categoria INT NOT NULL,
    nombre VARCHAR(100) NOT NULL,
    precio DECIMAL(10,2) NOT NULL DEFAULT 0.00,
    activo BOOLEAN NOT NULL DEFAULT TRUE,
    disponible BOOLEAN NOT NULL DEFAULT TRUE,

    CONSTRAINT fk_extras_categoria
        FOREIGN KEY (id_categoria)
        REFERENCES categorias(id_categoria),

    CONSTRAINT uq_extras_categoria_nombre
        UNIQUE (id_categoria, nombre),

    CONSTRAINT chk_extras_precio
        CHECK (precio >= 0)
);


CREATE TABLE IF NOT EXISTS detalles_pedido_extras (
    id_detalle_extra INT AUTO_INCREMENT PRIMARY KEY,
    id_detalle INT NOT NULL,
    id_extra INT NOT NULL,
    nombre VARCHAR(100) NOT NULL,
    cantidad INT NOT NULL DEFAULT 1,
    precio_unitario DECIMAL(10,2) NOT NULL,

    FOREIGN KEY (id_detalle)
        REFERENCES detalles_pedido(id_detalle)
        ON DELETE CASCADE,

    FOREIGN KEY (id_extra)
        REFERENCES extras(id_extra),

    CHECK (cantidad > 0),
    CHECK (precio_unitario >= 0)
);
